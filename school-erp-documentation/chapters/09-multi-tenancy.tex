\section{Safe Query Pattern}
\begin{lstlisting}[language=JavaScript]
records.filter((row) =>
  row.tenantId === scope.tenantId &&
  row.schoolId === scope.schoolId &&
  row.campusId === scope.campusId &&
  row.academicYearId === scope.academicYearId
)
\end{lstlisting}

\section{Unsafe Query Pattern}
\begin{lstlisting}[language=JavaScript]
// Unsafe: tenantId is accepted from a browser form.
const tenantId = formData.get("tenantId");
repository.findMany({ where: { tenantId } });
\end{lstlisting}

\begin{importantbox}
Tenant resolution must happen before domain service operations. UI state may choose filters, but it must not establish security authority.
\end{importantbox}

\textbf{Permission} & \textbf{Purpose}\\
\midrule

\texttt{communication.view/send/broadcast/notice.create/manage} & Communication Center capabilities.\\
\bottomrule

\section{Roles}
\begin{itemize}
\item Principal and system-admin have broad permissions.
\item Teacher has student, academic, attendance, and communication permissions.
\item Accountant has student, guardian, attendance, reporting, fees, and communication permissions.
\item HR has dashboard, HR, attendance mark, and communication permissions.
\end{itemize}
